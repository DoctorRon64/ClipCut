cmake_minimum_required(VERSION 3.25)

project(ClipCut LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

set(CLIP_OUTPUT_DIR "${CMAKE_BINARY_DIR}/bin")
set(CLIP_LIBRARY_DIR "${CMAKE_BINARY_DIR}/lib")

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CLIP_OUTPUT_DIR}")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CLIP_OUTPUT_DIR}")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CLIP_LIBRARY_DIR}")

add_subdirectory(libs/SDL)
add_subdirectory(ClipCut)
